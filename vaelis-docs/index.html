Vaelis is a 2D game engine with a visual editor, JavaScript scripting, Rapier2D physics, navigation, lighting and one-click export. These docs cover every component, API and editor control.

Build your first game Browse the docs
js
function onStart() {
  this.speed = 220;
}

function onUpdate(dt) {
  if (input.keyDown("ArrowRight")) {
    this.x += this.speed * dt;
  }
  if (input.keyPressed("Space")) {
    this.rigidbody.addImpulse(0, -350);
  }
}
